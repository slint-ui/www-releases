4/// Fill a rectangle of the frame buffer, made of packed RGB triplets, with a
5/// single color, composited over what the buffer already holds: a translucent
6/// color lets the pixels underneath show through, and a fully transparent one
7/// paints nothing. The rectangle is clipped to the buffer, whose length must
8/// be `buffer_size[0] * buffer_size[1] * 3`.
9pub fn fill_rect(
10    frame_buffer: &mut [u8],
11    buffer_size: [u32; 2],
12    position: [i32; 2],
13    size: [i32; 2],
14    color: crate::Color,
15) {
16    let alpha = color.alpha();
17    // A fully transparent color leaves every pixel as it was
18    if alpha == 0 {
19        return;
20    }
21    // The drawn span on each axis is the intersection of
22    // [position, position + size) with [0, buffer_size): a rectangle sticking
23    // out of the buffer, including one at a negative position, is shortened
24    let x0 = position[0].clamp(0, buffer_size[0] as i32) as usize;
25    let x1 = position[0].saturating_add(size[0]).clamp(0, buffer_size[0] as i32) as usize;
26    let y0 = position[1].clamp(0, buffer_size[1] as i32) as usize;
27    let y1 = position[1].saturating_add(size[1]).clamp(0, buffer_size[1] as i32) as usize;
28    let stride = buffer_size[0] as usize * 3;
29    let rgb = [color.red(), color.green(), color.blue()];
30    for row in y0..y1 {
31        let row_range = row * stride + x0 * 3..row * stride + x1 * 3;
32        let pixels = frame_buffer[row_range].as_chunks_mut::<3>().0;
33        if alpha == 0xff {
34            pixels.fill(rgb);
35        } else {
36            for pixel in pixels {
37                // The buffer carries no alpha channel, so the destination is
38                // opaque by construction, and the result is opaque in turn
39                let destination = crate::Color::from_rgb_u8(pixel[0], pixel[1], pixel[2]);
40                let blended = color.composite_over(destination);
41                *pixel = [blended.red(), blended.green(), blended.blue()];
42            }
43        }
44    }
45}

47#[test]
48fn test_fill_rect_negative_position() {
49    // A 3x2 rectangle at (-2, -1) intersects a 4x4 buffer in the single
50    // pixel (0, 0)
51    let mut buffer = [0u8; 4 * 4 * 3];
52    fill_rect(&mut buffer, [4, 4], [-2, -1], [3, 2], crate::Color::from_rgb_u8(1, 2, 3));
53    for y in 0..4 {
54        for x in 0..4 {
55            let expected = if (x, y) == (0, 0) { [1, 2, 3] } else { [0, 0, 0] };
56            assert_eq!(buffer[(y * 4 + x) * 3..][..3], expected, "pixel ({x}, {y})");
57        }
58    }
59
60    // A rectangle entirely outside the buffer paints nothing
61    let mut buffer = [7u8; 4 * 4 * 3];
62    fill_rect(&mut buffer, [4, 4], [-5, -5], [3, 2], crate::Color::from_rgb_u8(1, 2, 3));
63    assert_eq!(buffer, [7u8; 4 * 4 * 3]);
64}

66#[test]
67fn test_fill_rect_blends_with_the_destination() {
68    // A 2x2 rectangle of half-transparent red over a buffer of a single color
69    let destination = [200, 100, 50];
70    let mut buffer = [0u8; 4 * 4 * 3];
71    buffer.as_chunks_mut::<3>().0.fill(destination);
72    fill_rect(&mut buffer, [4, 4], [1, 1], [2, 2], crate::Color::from_argb_encoded(0x80ff0000));
73    // 255 * 128 + 200 * 127 + 127 == 58167, and 58167 / 255 == 228; the green
74    // and blue channels keep 127/255 of the destination alone
75    let blended = [228, 50, 25];
76    for y in 0..4 {
77        for x in 0..4 {
78            let expected =
79                if (1..3).contains(&x) && (1..3).contains(&y) { blended } else { destination };
80            assert_eq!(buffer[(y * 4 + x) * 3..][..3], expected, "pixel ({x}, {y})");
81        }
82    }
83
84    // Blending is clipped like any other fill: at (-1, -1) only the pixel
85    // (0, 0) is blended
86    let mut buffer = [0u8; 4 * 4 * 3];
87    buffer.as_chunks_mut::<3>().0.fill(destination);
88    fill_rect(&mut buffer, [4, 4], [-1, -1], [2, 2], crate::Color::from_argb_encoded(0x80ff0000));
89    for y in 0..4 {
90        for x in 0..4 {
91            let expected = if (x, y) == (0, 0) { blended } else { destination };
92            assert_eq!(buffer[(y * 4 + x) * 3..][..3], expected, "pixel ({x}, {y})");
93        }
94    }
95}

97#[test]
98fn test_fill_rect_transparent() {
99    // A fully transparent color paints nothing, whatever its other channels
100    let mut buffer = [7u8; 4 * 4 * 3];
101    fill_rect(&mut buffer, [4, 4], [0, 0], [4, 4], crate::Color::from_argb_encoded(0x00ff8040));
102    assert_eq!(buffer, [7u8; 4 * 4 * 3]);
103    // The default color is transparent
104    fill_rect(&mut buffer, [4, 4], [0, 0], [4, 4], crate::Color::default());
105    assert_eq!(buffer, [7u8; 4 * 4 * 3]);
106}
